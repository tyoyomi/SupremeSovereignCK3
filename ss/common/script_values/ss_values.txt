ss_perk_gift_points_value = {
	value = 0
	if = {
		limit = {
			exists = var:ss_cfg_perk_gift
			var:ss_cfg_perk_gift = 1
		}
		value = 1
	}
	else_if = {
		limit = {
			exists = var:ss_cfg_perk_gift
			var:ss_cfg_perk_gift = 2

		limit = {
			exists = var:ss_cfg_perk_gift

	}
}
