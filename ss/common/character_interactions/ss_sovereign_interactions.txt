# Supreme Sovereign — grant to direct child (Option A)
# Craft shape reused from Exalted Advisor bestow interaction.
# AI fully off: no ai_targets (chance must be (0-1]; omit when never AI).

@ss_grant_prestige_cost = 500

ss_grant_supreme_sovereign_interaction = {
	category = interaction_category_friendly
	icon = council
	interface_priority = 70
	common_interaction = yes

	desc = ss_grant_supreme_sovereign_interaction_desc

	auto_accept = yes

	cooldown = { days = 365 }
	cooldown_against_recipient = { days = 3650 }

	is_shown = {
		scope:actor = {
			is_alive = yes
			is_ai = no
			has_trait = Supreme_Sovereign
		}
		scope:recipient = {
			is_alive = yes
			NOT = { has_trait = Supreme_Sovereign }
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			custom_tooltip = {
				text = ss_tt_actor_may_grant
				ss_can_actor_grant_sovereign = yes
			}
			prestige >= @ss_grant_prestige_cost
		}
		scope:recipient = {
			custom_tooltip = {
				text = ss_tt_can_receive_sovereign
				ss_can_receive_sovereign = yes
			}
		}
	}

	cost = {
		prestige = @ss_grant_prestige_cost
	}

	on_accept = {
		scope:recipient = {
			ss_grant_supreme_sovereign_effect = yes
		}
	}

	ai_frequency = 0
	ai_potential = {
		always = no
	}
	ai_will_do = {
		base = 0
	}
}
