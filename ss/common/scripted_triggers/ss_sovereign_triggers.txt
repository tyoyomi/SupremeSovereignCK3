	has_trait = Supreme_Sovereign

	is_alive = yes
	is_ai = no

	is_alive = yes

# Faith Conquest — defender faith is hostile or evil toward attacker (no holy-war doctrine stack)

	scope:attacker.faith = {
		faith_hostility_level = {

		}
	}

# --- Distribute titles to children (CM-35) ---
# Soft peer: SU missing ⇒ has_variable fails ⇒ pending = no (always pass).

ss_su_unity_choice_pending_on_self = {
	has_variable = su_unity_choice_pending
}

# this = title; root = actor
# DTC15: never distribute titles at/above actor primary tier (avoids independent peers)
ss_is_distributable_title = {
	exists = holder
	holder = root
	exists = root.primary_title
	tier < root.primary_title.tier
	NOR = {
		this = root.primary_title
		AND = {
			exists = root.capital_county
			this = root.capital_county
		}
		AND = {
			tier = tier_barony
			exists = county
			exists = root.capital_county
			county = root.capital_county
		}
	}
}

# this = title already saved; root / scope:ss_dist_actor = actor
ss_title_is_dist_queued = {
	exists = scope:ss_dist_actor
	scope:ss_dist_actor = {
		is_target_in_variable_list = {
			name = ss_dist_queued_titles
			target = prev
		}
	}
}

# this = child; root = actor
ss_eligible_distribute_child = {
	is_alive = yes
	NOT = { this = root }
	is_child_of = root
	OR = {
		can_be_granted_titles_by = { RULER = root }
		can_be_granted_theocratic_titles_by = { RULER = root }
	}
	NOR = {
		has_trait = gallivanter
		AND = {
			has_trait = devoted
			root = { NOT = { government_has_flag = government_is_theocracy } }
		}
	}
}

ss_can_distribute_titles_to_children = {
	is_alive = yes
	is_ai = no
	has_trait = Supreme_Sovereign
	is_landed = yes
	NOT = { government_has_flag = government_is_landless_adventurer }
	exists = primary_title
	exists = capital_county
	NOT = { ss_su_unity_choice_pending_on_self = yes }
	any_child = {
		ss_eligible_distribute_child = yes
	}
	any_held_title = {
		tier >= tier_county
		ss_is_distributable_title = yes
	}
}

# --- Reclaim titles from children (CM-41) ---
# Invert distribute: take non-floor titles from living children (incl. independent peers).
# Soft peer: reuse ss_su_unity_choice_pending_on_self (missing SU = pass).

# this = title; root = actor
# Reclaim pool: personally held by a living child; not that child's capital county;
# baronies ride with counties (skip all baronies + capital-barony hygiene).
ss_is_reclaimable_child_title = {
	exists = holder
	holder = {
		is_alive = yes
		NOT = { this = root }
		is_child_of = root
	}
	tier >= tier_county
	NOR = {
		AND = {
			exists = holder.capital_county
			this = holder.capital_county
		}
		is_capital_barony = yes
	}
}

# this = child; root = actor
ss_eligible_reclaim_child = {
	is_alive = yes
	NOT = { this = root }
	is_child_of = root
	any_held_title = {
		ss_is_reclaimable_child_title = yes
	}
}

ss_can_reclaim_titles_from_children = {
	is_alive = yes
	is_ai = no
	has_trait = Supreme_Sovereign
	is_landed = yes
	NOT = { government_has_flag = government_is_landless_adventurer }
	exists = primary_title
	NOT = { ss_su_unity_choice_pending_on_self = yes }
	any_child = {
		ss_eligible_reclaim_child = yes
	}
}

# this = living character — qualify for set_house_head on own house
ss_can_reassert_house_head = {
	is_alive = yes
	has_trait = Supreme_Sovereign
	ss_permanent_house_head_enabled = yes
	exists = house
	is_house_head = no
	# PHH9: do not steal from another living Sovereign who already heads the house
	NAND = {
		exists = house.house_head
		house.house_head = {
			is_alive = yes
			has_trait = Supreme_Sovereign
		}
	}
}
